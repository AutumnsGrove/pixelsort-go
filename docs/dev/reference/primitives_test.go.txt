func TestRNGVectors(t *testing.T) {
	r := NewRNG(0)
	for i, want := range []uint64{0xe220a8397b1dcdaf, 0x6e789e6aa1b965f4, 0x06c45d188009454f} {
		if got := r.Uint64(); got != want {
			t.Fatalf("#%d got %#x want %#x", i, got, want)
		}
	}
	s := NewStream(1, 0)
	if a, b := s.Uint64(), s.Uint64(); a != 0xce30761cd7373f6d || b != 0xa82740738736e4c9 {
		t.Fatalf("stream(1,0) %#x %#x", a, b)
	}
	s = NewStream(7, 3)
	for i, want := range []uint32{74, 71, 14, 15, 14} {
		if got := s.Uint32n(100); got != want {
			t.Fatalf("Uint32n #%d got %d want %d", i, got, want)
		}
	}
}

func TestKeys(t *testing.T) {
	for _, c := range []struct{ r, g, b, hue, sat, luma int }{{255, 0, 0, 0, 255, 76}, {0, 255, 0, 85, 255, 149}, {0, 0, 255, 170, 255, 29}, {255, 255, 0, 42, 255, 225}, {0, 255, 255, 127, 255, 178}, {255, 0, 255, 212, 255, 105}, {128, 128, 128, 0, 0, 128}, {10, 200, 30, 89, 242, 123}, {0, 0, 0, 0, 0, 0}, {255, 255, 255, 0, 0, 255}, {200, 100, 50, 14, 191, 124}} {
		if hue(c.r, c.g, c.b) != c.hue || saturation(c.r, c.g, c.b) != c.sat || luma(c.r, c.g, c.b) != c.luma {
			t.Errorf("%v: hue %d sat %d luma %d", c, hue(c.r, c.g, c.b), saturation(c.r, c.g, c.b), luma(c.r, c.g, c.b))
		}
	}
}
func TestPartitionBijection(t *testing.T) {
	for _, sz := range [][2]int{{1, 1}, {1, 40}, {40, 1}, {2, 2}, {3, 5}, {97, 61}, {53, 53}, {61, 97}, {7, 7}, {64, 64}} {
		w, h := sz[0], sz[1]
		for a := 0; a < 360; a++ {
			L := Partition(w, h, a)
			if len(L.P) != w*h || int(L.Starts[len(L.Starts)-1]) != w*h {
				t.Fatalf("%dx%d a=%d: %d pixels", w, h, a, len(L.P))
			}
			seen := make([]bool, w*h)
			for _, p := range L.P {
				if seen[p] {
					t.Fatalf("%dx%d a=%d: duplicate", w, h, a)
				}
				seen[p] = true
			}
		}
	}
}
func TestCardinalDirections(t *testing.T) {
	first := func(a int) [][2]int {
		L := Partition(5, 3, a)
		var o [][2]int
		for _, p := range L.P[L.Starts[0]:L.Starts[1]] {
			o = append(o, [2]int{int(p) % 5, int(p) / 5})
		}
		return o
	}
	for a, want := range map[int][][2]int{0: {{0, 0}, {1, 0}, {2, 0}, {3, 0}, {4, 0}}, 90: {{0, 2}, {0, 1}, {0, 0}}, 180: {{4, 0}, {3, 0}, {2, 0}, {1, 0}, {0, 0}}, 270: {{0, 0}, {0, 1}, {0, 2}}} {
		got := first(a)
		if len(got) != len(want) {
			t.Fatalf("a=%d got %v want %v", a, got, want)
		}
		for i := range got {
			if got[i] != want[i] {
				t.Fatalf("a=%d got %v want %v", a, got, want)
			}
		}
	}
}
